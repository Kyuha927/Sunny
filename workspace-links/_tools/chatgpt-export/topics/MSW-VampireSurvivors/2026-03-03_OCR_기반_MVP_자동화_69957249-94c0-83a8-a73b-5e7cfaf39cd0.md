def _controller_focus_anchor(controller: Any, x: int, y: int) -> None:
    for method_name in ("right_click", "_right_click"):
        fn = getattr(controller, method_name, None)
        if callable(fn):
            fn(x, y)
            return


def _ocr_click_with_retry(controller: Any, goal: str, attempts: int = 3) -> Tuple[bool, str]:
    if attempts < 1:
        attempts = 1

    click_fn = getattr(controller, "_ocr_click_menu", None)
    if not callable(click_fn):
        return False, "controller missing _ocr_click_menu(goal, ...)"

    last_error = ""
    for idx in range(1, attempts + 1):
        try:
            ok = click_fn(goal)
            if ok:
                return True, ""
            last_error = f"goal={goal} attempt={idx} returned false"
        except Exception as exc:
            last_error = f"goal={goal} attempt={idx} exception={exc}"
        time.sleep(0.25)
    return False, last_error


def run_ocr_workflow(args: argparse.Namespace) -> Tuple[Optional[Any], Optional[str]]:
    if not args.ocr:
        return None, None

    try:
        from automation.msw_controller import MSWController, MYDESK_X, MYDESK_Y
    except Exception as exc:
        return None, f"unable to import automation.msw_controller: {exc}"

    controller: Any = None
    try:
        controller = MSWController(anchor_xy=(MYDESK_X, MYDESK_Y))
        _controller_focus_anchor(controller, MYDESK_X, MYDESK_Y)

        ok_play, reason_play = _ocr_click_with_retry(controller, "Play", attempts=3)
        if not ok_play:
            return controller, f"OCR_FAILED goal=Play reason={reason_play}"

        ok_start, reason_start = _ocr_click_with_retry(controller, "Start", attempts=3)
        if not ok_start:
            return controller, f"OCR_FAILED goal=Start reason={reason_start}"

        time.sleep(1.0)
        if _controller_interrupted(controller):
            return controller, "interrupted"
        return controller, None
    except Exception as exc:
        return controller, f"OCR_FAILED exception={exc}"

def _emit_checked_roots_debug(enabled: bool, checked_roots: List[Tuple[Path, bool]]) -> None:
    if not enabled:
        return
    p0 = sys.path[0] if len(sys.path) > 0 else ""
    p1 = sys.path[1] if len(sys.path) > 1 else ""
    p2 = sys.path[2] if len(sys.path) > 2 else ""
    checked = "; ".join(f"{root} exists={exists}" for root, exists in checked_roots)
    print(f"[ACCEPT][IMPORT] checked_roots=[{checked}] path0={p0} path1={p1} path2={p2}")


def _load_ocr_controller_symbols(
    debug_import: bool,
    ocr_controller_root: Optional[Path],
) -> Tuple[Optional[Any], Optional[int], Optional[int], Optional[str], bool]:
    repo_root = _ensure_repo_root_on_sys_path()
    candidate_roots: List[Path] = []
    if ocr_controller_root is not None:
        candidate_roots.append(ocr_controller_root.expanduser().resolve())
    if repo_root not in candidate_roots:
        candidate_roots.append(repo_root)

    checked_roots: List[Tuple[Path, bool]] = []
    import_errors: List[str] = []
    any_controller_file_found = False

    for root in candidate_roots:
        controller_path = root / CONTROLLER_REL_PATH
        exists = controller_path.exists()
        checked_roots.append((root, exists))
        _prepend_sys_path(root)

        if not exists:
            _emit_import_debug(debug_import, root, False, f"missing:{controller_path}")
            continue

        any_controller_file_found = True
        cls, ax, ay, err = _try_import_ocr_symbols(controller_path, debug_import, root)
        if err is None:
            return cls, ax, ay, None, False
        import_errors.append(err)

    _emit_checked_roots_debug(debug_import, checked_roots)

    if not any_controller_file_found:
