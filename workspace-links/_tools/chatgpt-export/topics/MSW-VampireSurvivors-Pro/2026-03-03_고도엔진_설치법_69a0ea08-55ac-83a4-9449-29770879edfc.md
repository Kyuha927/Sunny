## 11) `src/scenes/game.lua` (게임: ESC → 메뉴, 간단 이동)

```lua
local State = require("src.core.state")
local Input = require("src.core.input")

local Game = {}
Game.__index = Game

function Game.new(app)
  return setmetatable({ app = app, name = "game" }, Game)
end

function Game:update(dt)
  local p = State.player
  local dx, dy = 0, 0
  if Input.down("a") then dx = dx - 1 end
  if Input.down("d") then dx = dx + 1 end
  if Input.down("w") then dy = dy - 1 end
  if Input.down("s") then dy = dy + 1 end

  local len = math.sqrt(dx*dx + dy*dy)
  if len > 0 then
    dx, dy = dx/len, dy/len
    p.x = p.x + dx * p.speed * dt
    p.y = p.y + dy * p.speed * dt
  end
end

function Game:draw()
  local p = State.player
  love.graphics.print("GAME (WASD move, ESC menu)", 20, 20)
  love.graphics.circle("fill", p.x, p.y, 16)
end

function Game:keypressed(key)
  if key == "escape" then
    self.app.scenes:switch(require("src.scenes.menu").new(self.app))
  end
end

return Game
```

# 1) Unity → LÖVE 2D 이주 시 주의점

## 1️⃣ 엔진 패러다임 충돌 (Scene/Prefab/Inspector가 없음) 💀
- Unity: 씬/프리팹/컴포넌트/Inspector 값 = 프로젝트의 절반  
- LÖVE: **모든 상태가 코드/데이터 파일**이어야 함

✅ 대책  
- “Inspector 값”을 전부 **`data/*.lua|json`**로 내리고, 로딩하는 구조로 바꾸기  
- “씬”은 **Scene Manager(상태 머신)**로 교체

---

## 2️⃣ 라이프사이클/이벤트 모델 차이 🔥
- Unity: `Awake/Start/Update/FixedUpdate` + 코루틴 + 메시지/이벤트  
- LÖVE: `love.load/update/draw` + 직접 만든 스케줄러

✅ 대책  
- 코루틴 의존 로직은 **타이머/스케줄러 시스템**으로 교체  
- FixedUpdate에 있던 물리는 **dt clamp + fixed step**로 재구성 (안 하면 프레임 드랍 때 게임이 망가짐)

---

## 3️⃣ UI/레이아웃이 “없다” 수준 💀
- Unity UI: 앵커, 레이아웃 그룹, 이벤트 시스템, 폰트 처리  
- LÖVE: 직접 그려야 함 (버튼, 포커스, 스크롤, 입력)

✅ 대책  
- UI는 “웹/유니티 그대로” 가져오지 말고 **UI를 최소로 재설계**  
- 필요하면 **즉시 UI 라이브러리 도입**(예: SUIT 같은 immediate-mode UI) 또는 자체 최소 UI

---

## 4️⃣ 애니메이션/타임라인/툴링 손실 💸
- Unity 애니메이션/타임라인/씬툴 = 생산성 덩어리  
- LÖVE: 직접 구현하거나 외부 툴로 굴려야 함

# 🚀 “단순 UI”용 AI 협업 템플릿 (최소 UI 킷)

## 1) `src/ui/ui.lua` (상태 + 버튼)

```lua
local UI = {}

UI.hot = nil
UI.active = nil

local function hit(x,y,w,h,mx,my)
  return mx>=x and mx<=x+w and my>=y and my<=y+h
end

function UI.begin()
  UI.hot = nil
end

function UI.button(id, x, y, w, h, label)
  local mx, my = love.mouse.getPosition()
  local over = hit(x,y,w,h,mx,my)

  if over then UI.hot = id end

  local clicked = false
  if UI.active == id then
    if not love.mouse.isDown(1) then
      if over and UI.hot == id then clicked = true end
      UI.active = nil
    end
  elseif over and love.mouse.isDown(1) then
    UI.active = id
  end

  -- draw
  local isHot = (UI.hot == id)
  local isActive = (UI.active == id)

  local pad = isActive and 2 or 0
  love.graphics.rectangle("line", x+pad, y+pad, w-pad*2, h-pad*2)
  if isHot then
    love.graphics.rectangle("fill", x+1, y+1, w-2, h-2)
    love.graphics.setColor(0,0,0,1)
    love.graphics.print(label, x+10, y+10)
    love.graphics.setColor(1,1,1,1)
  else
    love.graphics.print(label, x+10, y+10)
  end

  return clicked
end

function UI.end_() end

return UI
```

> 색 지정이 들어가긴 했는데, **AI 디버깅용 가시성** 때문에 최소로 넣었습니다. 원하면 색 없이도 버전 드립니다.

# 1) `src/ui/ui.lua` (버튼/토글/슬라이더)

```lua
local UI = {}

UI.hot = nil
UI.active = nil
UI.idCounter = 0

local function nextId(prefix)
  UI.idCounter = UI.idCounter + 1
  return prefix .. "_" .. tostring(UI.idCounter)
end

local function clamp(v, a, b)
  if v < a then return a end
  if v > b then return b end
  return v
end

local function hit(x,y,w,h,mx,my)
  return mx>=x and mx<=x+w and my>=y and my<=y+h
end

function UI.begin()
  UI.hot = nil
  UI.idCounter = 0
end

local function drawBox(x,y,w,h, isHot, isActive)
  -- 단색/라인만: 스타일 최소 (디버깅 우선)
  if isActive then
    love.graphics.rectangle("line", x+2, y+2, w-4, h-4)
  else
    love.graphics.rectangle("line", x, y, w, h)
  end
  if isHot then
    love.graphics.rectangle("line", x+1, y+1, w-2, h-2)
  end
end

function UI.button(label, x, y, w, h)
  local id = nextId("btn")
  local mx, my = love.mouse.getPosition()
  local over = hit(x,y,w,h,mx,my)

  if over then UI.hot = id end

  local clicked = false
  if UI.active == id then
    if not love.mouse.isDown(1) then
      if over and UI.hot == id then clicked = true end
      UI.active = nil
    end
  elseif over and love.mouse.isDown(1) then
    UI.active = id
  end

  local isHot = (UI.hot == id)
  local isActive = (UI.active == id)

  drawBox(x,y,w,h, isHot, isActive)
  love.graphics.print(label, x + 10, y + 10)

  return clicked
end

function UI.toggle(label, value, x, y, w, h)
  local id = nextId("tgl")
  local mx, my = love.mouse.getPosition()
  local over = hit(x,y,w,h,mx,my)

  if over then UI.hot = id end

  local changed = false
  if UI.active == id then
    if not love.mouse.isDown(1) then
      if over and UI.hot == id then
        value = not value
        changed = true
      end
      UI.active = nil
    end
  elseif over and love.mouse.isDown(1) then
    UI.active = id
  end

  local isHot = (UI.hot == id)
  local isActive = (UI.active == id)

  drawBox(x,y,w,h, isHot, isActive)
  love.graphics.print(label .. ": " .. (value and "ON" or "OFF"), x + 10, y + 10)

  return value, changed
end

function UI.slider(label, value, minV, maxV, x, y, w, h)
  local id = nextId("sld")
  local mx, my = love.mouse.getPosition()
  local over = hit(x,y,w,h,mx,my)

  if over then UI.hot = id end

  local changed = false
  if UI.active == id then
    if love.mouse.isDown(1) then
      local t = (mx - x) / w
      t = clamp(t, 0, 1)
      local newV = minV + t * (maxV - minV)
      if newV ~= value then
        value = newV
        changed = true
      end
    else
      UI.active = nil
    end
  elseif over and love.mouse.isDown(1) then
    UI.active = id
  end

  local isHot = (UI.hot == id)
  local isActive = (UI.active == id)
  drawBox(x,y,w,h, isHot, isActive)

  -- 핸들 표시(단순)
  local t = (value - minV) / (maxV - minV)
  t = clamp(t, 0, 1)
  local hx = x + t * w
  love.graphics.line(hx, y, hx, y+h)

  love.graphics.print(string.format("%s: %.0f", label, value), x + 10, y + 10)

  return value, changed
end

return UI
```

# 2) `src/scenes/settings.lua` (설정 씬)

```lua
local UI = require("src.ui.ui")
local State = require("src.core.state")
local MenuScene = require("src.scenes.menu")

local Settings = {}
Settings.__index = Settings

function Settings.new(app)
  return setmetatable({ app = app, name = "settings" }, Settings)
end

function Settings:draw()
  UI.begin()

  love.graphics.print("SETTINGS (ESC to menu)", 20, 20)

  -- Fullscreen 토글 (State에 저장)
  State.settings = State.settings or { fullscreen = false, volume = 80 }
  State.settings.fullscreen, _ = UI.toggle("Fullscreen", State.settings.fullscreen, 20, 80, 320, 48)

  -- Volume 슬라이더
  State.settings.volume, _ = UI.slider("Volume", State.settings.volume, 0, 100, 20, 140, 320, 48)

  if UI.button("Back", 20, 220, 180, 48) then
    self.app.scenes:switch(MenuScene.new(self.app))
  end
end

function Settings:keypressed(key)
  if key == "escape" then
    self.app.scenes:switch(MenuScene.new(self.app))
  end
end

return Settings
```

```md
## Combat + Dialog Vertical Slice SPEC

### Goal
- 전투(조작/이동/공격) 루프와 대화(텍스트/선택지) 루프를 하나의 프로젝트에서 충돌 없이 공존시킨다.
- 온라인/인앱/광고 없음. Steam PC 대상.

### Core Systems
- Scene: menu, settings, game
- In-game Mode(State.game.mode):
  - "combat": 전투/조작 가능
  - "dialog": 대화/선택지 UI만 가능 (전투 입력 차단)
  - "pause": 일시정지 (옵션)

### Invariants (절대 규칙)
- 입력은 Input 모듈을 통해서만 읽는다. (직접 love.keyboard/mouse 사용 금지)
- 모드는 State.game.mode 단일 값으로만 전환한다. (여러 플래그 금지)
- 대화 중에는 전투 입력/업데이트를 중단한다. (combat update는 mode=="combat"에서만)
- 대화 렌더는 UI 킷(src/ui/ui.lua)만 사용한다. (임의 클릭 판정 금지)
- 대화 데이터는 data/dialog/*.lua(또는 json)로 분리한다. 하드코딩 금지.

### Expected Behavior (검증 가능)
1) Game 씬 진입 시 기본은 combat 모드.
2) 이동(WASD) + 공격(예: Space) 동작.
3) 특정 키(예: E)로 샘플 대화 시작 → mode="dialog"
4) dialog 모드에서:
   - 텍스트 박스 표시
   - Next(또는 Space)로 다음 줄 진행
   - 선택지가 있으면 버튼으로 선택
5) 대화 종료 시 mode="combat" 복귀, 전투 재개.
```

```md
## Dialog Typewriter Effect SPEC

### Goal
- 대화 텍스트가 글자 단위로 점진적으로 표시된다.
- 사용자는 "가속/스킵"으로 즉시 전체 문장을 표시하거나 다음으로 넘어갈 수 있다.

### Invariants
- 원본 대사 문자열은 절대 변경하지 않는다.
```

```md
## Dialog Stabilization (Phase 2) SPEC

### Goal
- 대화 진행/선택지/씬 전환(open_settings)/모드 전환(pause)에서 상태 꼬임을 0에 가깝게 줄인다.

- 입력은 src/core/input.lua 경유.
- mode는 State.game.mode 단일 값.
- dialog 데이터는 data/dialog/*.lua만 사용.
- typewriter는 revealCount/revealTimer/isRevealed만 사용, 원문 text 변경 금지.
- UI는 src/ui/ui.lua만 사용.

### Dialog Runtime Contract
State.game.dialog (하나의 구조체만 사용)
- active: boolean
- id: string
- lines: table
- lineIndex: integer (1..#lines)
- ended: boolean
- revealCount: integer
- revealTimer: number
- isRevealed: boolean
- choiceIndex: integer (1..#choices) [choices 있을 때만]
- inputLockTimer: number (0.0~; 전환 직후 입력 씹힘 방지)

### Rules
1) start(id):
   - mode="dialog"
   - dialog.active=true, ended=false
   - lineIndex=1로 시작
   - reveal reset
   - choiceIndex=1 (choices 있을 때)
   - inputLockTimer=0.12 (프레임/입력 중복 방지)

2) onLineEnter():
   - reveal reset (revealCount=0, revealTimer=0, isRevealed=false)
   - choiceIndex clamp
   - inputLockTimer=0.08 (줄 넘어갈 때 연타 방지)

3) update(dt):
   - inputLockTimer > 0 이면 감소시키고, 입력 처리 무시
   - lineIndex clamp (1..#lines)
   - 현재 line에 text가 없으면 안전 종료(safeClose)
   - choices가 있는 줄:
     - isRevealed==true일 때만 선택 이동 허용
   - endDialog==true면 safeClose()

4) keypressed / input handling:
   - confirm(Enter/Space):
     a) typing 중(isRevealed==false) -> 즉시 완전 표시
     b) 완전 표시 상태에서 choices 있으면 -> 선택 확정(handleChoice)
     c) 완전 표시 상태에서 choices 없으면 -> nextLine()
   - up/down:
     - choices 있을 때 && isRevealed==true일 때만 choiceIndex 변경
   - cancel(Escape):
     - (정책) dialog에서는 기본적으로 무시하거나, 선택지 있는 경우에만 "뒤로" 허용 (프로젝트 정책으로 고정)

5) safeClose(reason):
   - State.game.dialog = nil
   - mode="combat"로 복귀 (또는 호출자가 원하면 menu/settings로 전환)
   - ended 플래그를 남기지 않는다(잔상 상태 금지)

6) Scene switch safety:
   - open_settings 같은 action으로 settings 씬으로 전환할 때,
```

# 왜 앱/브라우저에 따라 달라지나? (핵심만)
1) **전송 단축키가 다를 수 있음**
- 어떤 환경은 `Enter`가 전송
- 어떤 환경은 `Enter`가 줄바꿈, `Ctrl+Enter`가 전송(설정/입력창 상태에 따라)

2) **포커스/입력창 핸들링이 다름**
- 브라우저는 주소창/탭/확장프로그램 영향
- 데스크탑 앱(Electron)은 포커스가 비교적 안정적이지만, 업데이트/단축키 정책이 바뀔 수 있음

그래서 스크립트는 **“전송키를 설정으로 바꾸게”** 해야 디버깅 확률이 내려갑니다 💀

```ahk
}

  global threshold
  ; 1) 전송
  SendChatMessage()

  ; 2) 카운트 +1
  n := ReadCounter()
  n += 1
  WriteCounter(n)

    PrepareNewChat()

  }
```

```powershell
$spec

$check

$gitStatus

$gitLog
```

## `tools\handoff\handoff_hotkeys.ahk` (Chrome 고정)

```ahk
#Requires AutoHotkey v2.0
#SingleInstance Force

; ============== CONFIG (CHROME WEB) ==============
threshold := 30

; 전송 방식: Chrome에서 가장 안정적인 전송키
sendMode := "CtrlEnter"

; 핫키: 선생님 요청
hotkeySendAndCount := "PgDn"
hotkeyPrepareNewChat := "PgUp"
hotkeyResetCounter := "Home"

projectRoot := "C:\Users\jhk92\OneDrive\문서\GitHub\ai\Moltbot\love-ai-template"
handoffOut  := "C:\Users\jhk92\OneDrive\문서\GitHub\ai\Moltbot\HANDOFF_AUTO.md"
psScript    := "C:\Users\jhk92\OneDrive\문서\GitHub\ai\Moltbot\tools\handoff\make_handoff.ps1"

counterFile := "C:\Users\jhk92\OneDrive\문서\GitHub\ai\Moltbot\.chat_turn_counter.txt"
chatUrl := "https://chatgpt.com/"
chromeExe := "chrome.exe"
; ==================================================

ReadCounter() {
  global counterFile
  try {
    s := Trim(FileRead(counterFile, "UTF-8"))
    if (s = "")
      return 0
    return Integer(s)
  } catch {
    return 0
  }
}

WriteCounter(n) {
  global counterFile
  try FileDelete(counterFile)
  FileAppend(String(n), counterFile, "UTF-8")
}

SendChatMessage() {
  global sendMode
  ; ⚠️ 포커스가 메시지 입력창에 있어야 합니다.
  if (sendMode = "CtrlEnter") {
    Send("^{Enter}")
  } else {
    Send("{Enter}")
  }
}

FocusChrome() {
  global chromeExe
  ; Chrome 창을 앞으로
  if WinExist("ahk_exe " chromeExe) {
    WinActivate("ahk_exe " chromeExe)
    Sleep(150)
    ; 주소창 포커스 빠져나오기(페이지로 포커스)
    Send("{Esc}")
    Sleep(50)
  }
}

PrepareNewChat() {
  global projectRoot, handoffOut, psScript, chatUrl

  ; HANDOFF 생성
  RunWait('powershell -NoProfile -ExecutionPolicy Bypass -File "' psScript '" -ProjectRoot "' projectRoot '" -OutFile "' handoffOut '"', , "Hide")

  ; HANDOFF 읽기
  handoff := FileRead(handoffOut, "UTF-8")

  ; 클립보드 주입
  A_Clipboard :=
(
"아래 HANDOFF 기준으로 이어서 진행해.
우선순위: Dialog 안정화 Phase 2.
Non-negotiables: Input via src/core/input.lua, mode via State.game.mode, dialog data external, typewriter reveal state only, UI via src/ui/ui.lua, settings only State.settings.

[HANDOFF]
" handoff
)

  ; 새 탭 열기(웹)
  Run(chatUrl)

  ; Chrome 포커스
  FocusChrome()

  TrayTip("New Chat Ready", "클립보드 복사 완료. 새 채팅에서 Ctrl+V -> Enter", 3)
}

SendAndCount() {
  global threshold
  ; 전송
  SendChatMessage()

  ; 카운트
  n := ReadCounter()
  n += 1
  WriteCounter(n)

  ; 임계치 도달 시 새 채팅 세팅
  if (n >= threshold) {
    TrayTip("Threshold hit", "턴 " n " / " threshold " → 새 채팅 세팅 실행", 2)
    PrepareNewChat()
  } else {
    TrayTip("Turn Count", "턴 " n " / " threshold, 1)
  }
}

ResetCounter() {
  WriteCounter(0)
  TrayTip("Reset", "턴 카운터 리셋(0)", 2)
}

; ===== hotkey bind =====
Hotkey(hotkeySendAndCount, (*) => SendAndCount())
Hotkey(hotkeyPrepareNewChat, (*) => PrepareNewChat())
Hotkey(hotkeyResetCounter, (*) => ResetCounter())
```
