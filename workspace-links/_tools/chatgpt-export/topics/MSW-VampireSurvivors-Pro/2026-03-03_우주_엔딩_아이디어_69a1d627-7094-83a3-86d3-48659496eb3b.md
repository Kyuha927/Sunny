## ✅ 파일 1) `ending_sequence.lua` (그대로 새 파일로 저장)

```lua
-- ending_sequence.lua
-- Minimal, low-bug-risk ending sequence for LÖVE2D
-- Flow:
-- fireworks/earth zoom (static image) -> subtle window frame overlay -> crystal clink -> black -> title

local EndingSequence = {}
EndingSequence.__index = EndingSequence

local function clamp01(x) return math.max(0, math.min(1, x)) end

function EndingSequence.new(opts)
  opts = opts or {}
  local self = setmetatable({}, EndingSequence)

  -- Assets
  self.imgEarthFireworks = assert(opts.imgEarthFireworks, "imgEarthFireworks required (Image)")
  self.imgWindowOverlay  = assert(opts.imgWindowOverlay,  "imgWindowOverlay required (Image, can be 1px line)")
  self.sfxClink          = assert(opts.sfxClink,          "sfxClink required (Source)")

  -- Callbacks
  self.onDoneToTitle = assert(opts.onDoneToTitle, "onDoneToTitle callback required")

  -- Timing (seconds)
  self.t = 0
  self.T_EARTH     = opts.T_EARTH or 2.3  -- earth/fireworks hold
  self.T_OVERLAY   = opts.T_OVERLAY or 0.3 -- window hint duration
  self.T_CLINK_AT  = opts.T_CLINK_AT or 0.0 -- offset into overlay to play clink
  self.T_BLACK     = opts.T_BLACK or 0.5  -- black before title

  -- Internal
  self.state = "EARTH"  -- EARTH -> OVERLAY -> BLACK -> DONE
  self.stateT = 0
  self.playedClink = false

  -- Visual params
  self.earthZoomStart = opts.earthZoomStart or 1.0
  self.earthZoomEnd   = opts.earthZoomEnd   or 0.75 -- slight zoom-out feel (keep subtle, low risk)
  self.earthAlpha     = 1.0

  -- Overlay intensity (keep subtle)
  self.overlayAlpha = opts.overlayAlpha or 0.18  -- subtle!
  self.blackAlpha = 0

  return self
end

function EndingSequence:enter()
  self.t = 0
  self.state = "EARTH"
  self.stateT = 0
  self.playedClink = false
  self.blackAlpha = 0
end

function EndingSequence:update(dt)
  self.t = self.t + dt
  self.stateT = self.stateT + dt

  if self.state == "EARTH" then
    if self.stateT >= self.T_EARTH then
      self.state = "OVERLAY"
      self.stateT = 0
    end

  elseif self.state == "OVERLAY" then
    -- Play clink once at overlay start + offset
    if (not self.playedClink) and self.stateT >= self.T_CLINK_AT then
      self.playedClink = true
      self.sfxClink:stop()
      self.sfxClink:play()
    end

    if self.stateT >= self.T_OVERLAY then
      self.state = "BLACK"
      self.stateT = 0
      self.blackAlpha = 0
    end

  elseif self.state == "BLACK" then
    -- Fade to black quickly (but deterministic)
    local a = clamp01(self.stateT / math.max(self.T_BLACK, 0.001))
    self.blackAlpha = a

    if self.stateT >= self.T_BLACK then
      self.state = "DONE"
      self.stateT = 0
      -- Immediately go to title
      self.onDoneToTitle()
    end
  end
end

function EndingSequence:draw()
  local w, h = love.graphics.getWidth(), love.graphics.getHeight()

  -- EARTH image (fireworks over earth) with a subtle zoom-out
  local zoomT = 0
  if self.state == "EARTH" then
    zoomT = clamp01(self.stateT / math.max(self.T_EARTH, 0.001))
  else
    zoomT = 1
  end
  local scale = self.earthZoomStart + (self.earthZoomEnd - self.earthZoomStart) * zoomT

  local img = self.imgEarthFireworks
  local iw, ih = img:getWidth(), img:getHeight()

  love.graphics.push("all")
  love.graphics.origin()
  love.graphics.setColor(1, 1, 1, 1)

  -- Centered draw
  local cx, cy = w * 0.5, h * 0.5
  love.graphics.draw(img, cx, cy, 0, scale, scale, iw * 0.5, ih * 0.5)

  -- OVERLAY: subtle window hint (thin line / frame)
  if self.state == "OVERLAY" then
    love.graphics.setColor(1, 1, 1, self.overlayAlpha)
    local ov = self.imgWindowOverlay
    local ow, oh = ov:getWidth(), ov:getHeight()
    -- stretch overlay to screen (assumes 1x1 or designed for stretching)
    love.graphics.draw(ov, 0, 0, 0, w / ow, h / oh)
  end

  -- BLACK: full-screen black fade
  if self.state == "BLACK" then
    love.graphics.setColor(0, 0, 0, self.blackAlpha)
    love.graphics.rectangle("fill", 0, 0, w, h)
  end

  love.graphics.pop()
end

function EndingSequence:keypressed(key)
  -- Optional: allow skip? Default OFF to preserve impact.
  -- If you want skip in dev only, gate behind a flag.
end

return EndingSequence
```

```lua
local EndingSequence = require("ending_sequence")

local scene = "title"
local ending = nil

-- Assets (임시 경로)
local imgEarthFireworks
local imgWindowOverlay
local sfxClink

local function goTitle()
  scene = "title"
  ending = nil
end

local function startEnding()
  scene = "ending"
  ending = EndingSequence.new({
    imgEarthFireworks = imgEarthFireworks,
    imgWindowOverlay  = imgWindowOverlay,
    sfxClink          = sfxClink,
    onDoneToTitle     = goTitle,

    -- timing tuned to our decisions
    T_EARTH    = 2.3,
    T_OVERLAY  = 0.3,
    T_CLINK_AT = 0.0,
    T_BLACK    = 0.5,

    overlayAlpha = 0.18,     -- subtle
    earthZoomStart = 1.0,
    earthZoomEnd   = 0.75,
  })
  ending:enter()
end

function love.load()
  love.graphics.setDefaultFilter("nearest", "nearest")

  imgEarthFireworks = love.graphics.newImage("assets/ending/earth_fireworks.png")
  imgWindowOverlay  = love.graphics.newImage("assets/ending/window_overlay.png")
  sfxClink          = love.audio.newSource("assets/ending/crystal_clink.wav", "static")

  -- Dev: 바로 엔딩 보려면 이 줄 켜고 테스트
  -- startEnding()
end

function love.update(dt)
  if scene == "ending" and ending then
    ending:update(dt)
  end
end

function love.draw()
  if scene == "title" then
    love.graphics.print("TITLE (press E to test ending)", 20, 20)
  elseif scene == "ending" and ending then
    ending:draw()
  end
end

function love.keypressed(key)
  if scene == "title" then
    if key == "e" then startEnding() end
  elseif scene == "ending" and ending then
    ending:keypressed(key)
  end
end
```

# 1) `terrain.lua`

```lua
-- terrain.lua
-- Practical curve-terrain for LÖVE2D:
-- - Authoring as polyline control points
-- - Smooth render using Catmull-Rom sampling
-- - Collision as heightfield: y = groundY(x) from the polyline (safe & stable)
-- - Save/Load as Lua table (no JSON dependency)

local Terrain = {}
Terrain.__index = Terrain

local function clamp(x, a, b) return math.max(a, math.min(b, x)) end

local function dist2(ax, ay, bx, by)
  local dx, dy = ax - bx, ay - by
  return dx*dx + dy*dy
end

local function deepcopyPoints(points)
  local out = {}
  for i = 1, #points do
    out[i] = { x = points[i].x, y = points[i].y }
  end
  return out
end

-- Catmull-Rom interpolation (uniform)
local function catmullRom(p0, p1, p2, p3, t)
  local t2 = t * t
  local t3 = t2 * t
  local x =
      0.5 * ((2*p1.x) +
      (-p0.x + p2.x) * t +
      (2*p0.x - 5*p1.x + 4*p2.x - p3.x) * t2 +
      (-p0.x + 3*p1.x - 3*p2.x + p3.x) * t3)
  local y =
      0.5 * ((2*p1.y) +
      (-p0.y + p2.y) * t +
      (2*p0.y - 5*p1.y + 4*p2.y - p3.y) * t2 +
      (-p0.y + 3*p1.y - 3*p2.y + p3.y) * t3)
  return x, y
end

function Terrain.new()
  local self = setmetatable({}, Terrain)
  self.points = {}          -- control points (sorted by x)
  self._segments = nil      -- cached segments for fast sampling
  self._dirty = true
  self.renderSamples = 10   -- per segment (render smoothing)
  self.groundPaddingY = 2000 -- used if x is outside range
  return self
end

function Terrain:markDirty()
  self._dirty = true
  self._segments = nil
end

function Terrain:setPoints(points)
  self.points = deepcopyPoints(points or {})
  table.sort(self.points, function(a, b) return a.x < b.x end)
  self:markDirty()
end

function Terrain:getPoints()
  return deepcopyPoints(self.points)
end

function Terrain:addPoint(x, y)
  table.insert(self.points, { x = x, y = y })
  table.sort(self.points, function(a, b) return a.x < b.x end)
  self:markDirty()
end

function Terrain:removePointAt(index)
  if index and self.points[index] then
    table.remove(self.points, index)
    self:markDirty()
  end
end

function Terrain:movePoint(index, x, y, lockXSort)
  local p = self.points[index]
  if not p then return end
  p.x, p.y = x, y
  if not lockXSort then
    table.sort(self.points, function(a, b) return a.x < b.x end)
  end
  self:markDirty()
end

function Terrain:_rebuildSegmentsIfNeeded()
  if not self._dirty then return end
  self._dirty = false

  local pts = self.points
  local segs = {}

  -- Build monotonic x segments for collision sampling
  for i = 1, #pts - 1 do
    local a, b = pts[i], pts[i+1]
    if b.x ~= a.x then
      segs[#segs+1] = { x0 = a.x, y0 = a.y, x1 = b.x, y1 = b.y }
    end
  end

  table.sort(segs, function(s1, s2) return s1.x0 < s2.x0 end)
  self._segments = segs
end

-- Binary search segment by x
function Terrain:_findSegmentIndex(x)
  local segs = self._segments
  local lo, hi = 1, #segs
  while lo <= hi do
    local mid = (lo + hi) // 2
    local s = segs[mid]
    if x < s.x0 then
      hi = mid - 1
    elseif x > s.x1 then
      lo = mid + 1
    else
      return mid
    end
  end
  return nil
end

-- Collision-safe ground height: piecewise linear from control points.
-- (Rendering can be smooth; physics stays stable.)
function Terrain:groundY(x)
  self:_rebuildSegmentsIfNeeded()
  local segs = self._segments
  if #segs == 0 then return nil end

  -- Outside range: clamp to edge heights
  if x <= segs[1].x0 then return segs[1].y0 end
  if x >= segs[#segs].x1 then return segs[#segs].y1 end

  local idx = self:_findSegmentIndex(x)
  if not idx then return nil end
  local s = segs[idx]
  local t = (x - s.x0) / (s.x1 - s.x0)
  return s.y0 + (s.y1 - s.y0) * t
end

-- Optional: smooth curve samples for drawing only (Catmull-Rom)
function Terrain:buildRenderPolyline()
  local pts = self.points
  if #pts < 2 then return {} end

  local samples = {}
  local n = #pts
  local steps = self.renderSamples

  for i = 1, n - 1 do
    local p0 = pts[clamp(i-1, 1, n)]
    local p1 = pts[i]
    local p2 = pts[i+1]
    local p3 = pts[clamp(i+2, 1, n)]

    for s = 0, steps - 1 do
      local t = s / steps
      local x, y = catmullRom(p0, p1, p2, p3, t)
      samples[#samples+1] = x
      samples[#samples+1] = y
    end
  end

  -- add last point
  samples[#samples+1] = pts[n].x
  samples[#samples+1] = pts[n].y

  return samples
end

function Terrain:saveToFile(path)
  local pts = self.points
  local lines = {}
  lines[#lines+1] = "return { points = {"
  for i = 1, #pts do
    lines[#lines+1] = string.format("  { x = %.3f, y = %.3f },", pts[i].x, pts[i].y)
  end
  lines[#lines+1] = "} }"
  local ok, err = love.filesystem.write(path, table.concat(lines, "\n"))
  return ok, err
end

function Terrain:loadFromFile(path)
  if not love.filesystem.getInfo(path) then
    return false, "file not found: " .. tostring(path)
  end
  local chunk, err = love.filesystem.load(path)
  if not chunk then return false, err end
  local data = chunk()
  if type(data) ~= "table" or type(data.points) ~= "table" then
    return false, "invalid terrain file"
  end
  self:setPoints(data.points)
  return true
end

function Terrain:draw(mode, camX, camY)
  camX, camY = camX or 0, camY or 0
  local pts = self.points
  if #pts < 2 then return end

  love.graphics.push("all")
  love.graphics.translate(-camX, -camY)

  if mode == "smooth" then
    local poly = self:buildRenderPolyline()
    if #poly >= 4 then
      love.graphics.line(poly)
    end
  else
    -- raw polyline
    for i = 1, #pts - 1 do
      love.graphics.line(pts[i].x, pts[i].y, pts[i+1].x, pts[i+1].y)
    end
  end

  love.graphics.pop()
end

function Terrain:drawControlPoints(camX, camY, selectedIndex)
  camX, camY = camX or 0, camY or 0
  love.graphics.push("all")
  love.graphics.translate(-camX, -camY)

  for i = 1, #self.points do
    local p = self.points[i]
    if i == selectedIndex then
      love.graphics.circle("fill", p.x, p.y, 6)
    else
      love.graphics.circle("line", p.x, p.y, 5)
    end
  end

  love.graphics.pop()
end

function Terrain:pickPoint(x, y, camX, camY, radius)
  camX, camY = camX or 0, camY or 0
  radius = radius or 10
  local rr = radius * radius
  local wx, wy = x + camX, y + camY

  local best, bestD2 = nil, math.huge
  for i = 1, #self.points do
    local p = self.points[i]
    local d2 = dist2(wx, wy, p.x, p.y)
    if d2 <= rr and d2 < bestD2 then
      best, bestD2 = i, d2
    end
  end
  return best
end

return Terrain
```

```lua
-- editor.lua
-- Terrain editor controls:
-- LMB: add point (or drag if clicking a point)
-- RMB: delete nearest point
-- S: save, L: load, C: clear
-- G: toggle show ground samples debug
-- TAB: toggle edit mode (edit/play)

local Editor = {}
Editor.__index = Editor

function Editor.new(terrain)
  local self = setmetatable({}, Editor)
  self.terrain = terrain
  self.enabled = true
  self.selected = nil
  self.dragging = false
  self.lockSortWhileDrag = true -- avoids points reordering mid-drag (less surprising)
  self.filePath = "terrain_level1.lua"
  self.showDebugGround = false
  return self
end

function Editor:toggle()
  self.enabled = not self.enabled
  self.selected = nil
  self.dragging = false
end

function Editor:update(dt, camX, camY)
  -- nothing
end

function Editor:mousepressed(x, y, button, camX, camY)
  if not self.enabled then return end
  if button == 1 then
    local idx = self.terrain:pickPoint(x, y, camX, camY, 12)
    if idx then
      self.selected = idx
      self.dragging = true
    else
      self.terrain:addPoint(x + camX, y + camY)
      -- auto-select the closest newly added (best effort)
      self.selected = self.terrain:pickPoint(x, y, camX, camY, 14)
    end
  elseif button == 2 then
    local idx = self.terrain:pickPoint(x, y, camX, camY, 14)
    if idx then
      self.terrain:removePointAt(idx)
      self.selected = nil
      self.dragging = false
    end
  end
end

function Editor:mousereleased(x, y, button)
  if not self.enabled then return end
  if button == 1 then
    self.dragging = false
  end
end

function Editor:mousemoved(x, y, dx, dy, camX, camY)
  if not self.enabled then return end
  if self.dragging and self.selected then
    self.terrain:movePoint(self.selected, x + camX, y + camY, self.lockSortWhileDrag)
  end
end

function Editor:keypressed(key)
  if key == "tab" then
    self:toggle()
    return
  end
  if not self.enabled then return end

  if key == "s" then
    self.terrain:saveToFile(self.filePath)
  elseif key == "l" then
    self.terrain:loadFromFile(self.filePath)
  elseif key == "c" then
    self.terrain:setPoints({})
    self.selected = nil
  elseif key == "g" then
    self.showDebugGround = not self.showDebugGround
  end
end

function Editor:drawHUD()
  local mode = self.enabled and "EDIT" or "PLAY"
  love.graphics.print(
    ("[%s] LMB add/drag | RMB delete | TAB toggle | S save | L load | C clear | G debug"):format(mode),
    16, 16
  )
  if self.enabled then
    love.graphics.print(("file: %s"):format(self.filePath), 16, 36)
  end
end

return Editor
```

# 3) `player.lua`

```lua
-- player.lua
-- Safe movement on curve terrain:
-- - Gravity + jump
-- - Ground collision via terrain:groundY(x)
-- - Snap-to-ground (stable), ignores slope physics (reduces bugs)
-- This is "combat-safe baseline" even on curves.

local Player = {}
Player.__index = Player

function Player.new(x, y)
  local self = setmetatable({}, Player)
  self.x, self.y = x or 200, y or 200
  self.vx, self.vy = 0, 0
  self.w, self.h = 22, 34
  self.speed = 220
  self.jumpVel = -420
  self.gravity = 1200
  self.onGround = false

  -- Small vertical tolerance for landing (prevents jitter)
  self.groundEpsilon = 2
  return self
end

function Player:update(dt, terrain)
  local left  = love.keyboard.isDown("a") or love.keyboard.isDown("left")
  local right = love.keyboard.isDown("d") or love.keyboard.isDown("right")

  local ax = 0
  if left then ax = ax - 1 end
  if right then ax = ax + 1 end

  self.vx = ax * self.speed
  self.vy = self.vy + self.gravity * dt

  -- integrate
  self.x = self.x + self.vx * dt
  self.y = self.y + self.vy * dt

  -- ground collision (height at x)
  local groundY = terrain:groundY(self.x)
  if groundY then
    local footY = self.y + self.h * 0.5
    local targetFootY = groundY
    if footY >= targetFootY - self.groundEpsilon and self.vy >= 0 then
      -- land / snap
      self.onGround = true
      self.vy = 0
      self.y = (targetFootY - self.h * 0.5)
    else
      self.onGround = false
    end
  else
    self.onGround = false
  end
end

function Player:jump()
  if self.onGround then
    self.vy = self.jumpVel
    self.onGround = false
  end
end

function Player:draw(camX, camY)
  camX, camY = camX or 0, camY or 0
  love.graphics.push("all")
  love.graphics.translate(-camX, -camY)
  love.graphics.rectangle("line", self.x - self.w*0.5, self.y - self.h*0.5, self.w, self.h)
  love.graphics.pop()
end

return Player
```

# 4) `main.lua`

```lua
local Terrain = require("terrain")
local Editor  = require("editor")
local Player  = require("player")

local terrain
local editor
local player

-- simple camera following player
local camX, camY = 0, 0

function love.load()
  love.graphics.setDefaultFilter("nearest", "nearest")
  terrain = Terrain.new()
  editor = Editor.new(terrain)
  player = Player.new(240, 120)

  -- starter points (so you see a curve immediately)
  terrain:setPoints({
    {x = 50,  y = 420},
    {x = 220, y = 360},
    {x = 420, y = 400},
    {x = 680, y = 320},
    {x = 980, y = 420},
  })
end

function love.update(dt)
  -- Camera follows player (x only mostly)
  camX = player.x - love.graphics.getWidth() * 0.35
  camY = 0

  if not editor.enabled then
    player:update(dt, terrain)
  end
end

function love.draw()
  -- terrain
  terrain:draw("smooth", camX, camY)
  terrain:drawControlPoints(camX, camY, editor.selected)

  -- debug ground samples (optional)
  if editor.showDebugGround then
    love.graphics.push("all")
    love.graphics.translate(-camX, -camY)
    local w = love.graphics.getWidth()
    for sx = 0, w, 16 do
      local wx = sx + camX
      local gy = terrain:groundY(wx)
      if gy then
        love.graphics.points(wx, gy)
      end
    end
    love.graphics.pop()
  end

  -- player
  player:draw(camX, camY)

  -- HUD
  editor:drawHUD()
end

function love.keypressed(key)
  editor:keypressed(key)
  if not editor.enabled then
    if key == "space" or key == "w" or key == "up" then
      player:jump()
    end
  end
end

function love.mousepressed(x, y, button)
  editor:mousepressed(x, y, button, camX, camY)
end

function love.mousereleased(x, y, button)
  editor:mousereleased(x, y, button)
end

function love.mousemoved(x, y, dx, dy)
  editor:mousemoved(x, y, dx, dy, camX, camY)
end
```

## (B) 스냅샷/복원 유틸 추가 (editor.lua 상단 근처)

```lua
local function clonePoints(points)
  local out = {}
  for i=1,#points do
    local p = points[i]
    out[i] = { x=p.x, y=p.y, mat=p.mat }
  end
  return out
end

function Editor:_pushUndo(label)
  -- 현재 상태 스냅샷
  local snap = { label = label or "edit", points = clonePoints(self.terrain:getPoints()) }
  table.insert(self.undo, snap)
  if #self.undo > self.undoMax then table.remove(self.undo, 1) end
  -- 새 편집이 발생하면 redo는 날림
  self.redo = {}
end

function Editor:_applySnapshot(snap)
  self.terrain:setPoints(snap.points)
  self.selected = nil
  self.dragging = false
end

function Editor:undoOnce()
  if #self.undo == 0 then return end
  local current = { label="current", points = clonePoints(self.terrain:getPoints()) }
  table.insert(self.redo, current)

  local snap = table.remove(self.undo)
  self:_applySnapshot(snap)
end

function Editor:redoOnce()
  if #self.redo == 0 then return end
  local current = { label="current", points = clonePoints(self.terrain:getPoints()) }
  table.insert(self.undo, current)

  local snap = table.remove(self.redo)
  self:_applySnapshot(snap)
end
```

예: 선택 포인트가 없으면 “가장 가까운 세그먼트의 왼쪽 포인트”에 mat를 박기.

이건 세그먼트 피킹 함수가 필요합니다. (짧게 추가)

#### 세그먼트 피킹(간단 버전) — editor.lua에 추가
```lua
local function pointSegmentDist2(px, py, ax, ay, bx, by)
  local vx, vy = bx-ax, by-ay
  local wx, wy = px-ax, py-ay
  local c1 = vx*wx + vy*wy
  if c1 <= 0 then return (px-ax)^2 + (py-ay)^2 end
  local c2 = vx*vx + vy*vy
  if c2 <= c1 then return (px-bx)^2 + (py-by)^2 end
  local t = c1 / c2
  local projx, projy = ax + t*vx, ay + t*vy
  return (px-projx)^2 + (py-projy)^2
end

function Editor:pickSegment(x, y, camX, camY, radius)
  local pts = self.terrain:getPoints()
  if #pts < 2 then return nil end
  local wx, wy = x + camX, y + camY
  local rr = (radius or 16)^2
  local bestI, bestD2 = nil, math.huge
  for i=1,#pts-1 do
    local a,b = pts[i], pts[i+1]
    local d2 = pointSegmentDist2(wx, wy, a.x,a.y, b.x,b.y)
    if d2 <= rr and d2 < bestD2 then
      bestD2, bestI = d2, i
    end
  end
  return bestI
end
```

그리고 숫자키 처리에서 선택 포인트가 없을 때:


                Debug.LogWarning("LevelEditorTool: 씬에 LevelGenerator가 없거나 Level_1A 오브젝트를 찾을 수 없습니다. Level_1A에 LevelGenerator를 붙여 주세요.");
                return;
            }

            if (generator.groundPrefab == null)
            {
                Debug.LogWarning("LevelEditorTool: LevelGenerator의 Ground Prefab이 비어 있습니다.");
                return;
            }

            Vector3 worldPos = GetWorldPositionFromMouse(Event.current.mousePosition, sceneView);
            if (worldPos == Vector3.positiveInfinity) return;

            GameObject group = generator.GenerateStage1ASet(worldPos);
            if (group != null)
            {
                Undo.RegisterCreatedObjectUndo(group, "Spawn Stage 1-A Set");
                EditorUtility.SetDirty(generator.gameObject);
                if (generator.gameObject.scene.IsValid())
                    EditorSceneManager.MarkSceneDirty(generator.gameObject.scene);
            }

            Event.current.Use();
        }
    }

    private static LevelGenerator FindLevelGenerator()
    {
        GameObject levelRoot = GameObject.Find(LevelParentName);
        if (levelRoot != null)
        {
            var gen = levelRoot.GetComponent<LevelGenerator>();
            if (gen != null) return gen;
        }

        return Object.FindFirstObjectByType<LevelGenerator>(FindObjectsInactive.Include);
    }

    private static Vector3 GetWorldPositionFromMouse(Vector2 mousePosition, SceneView sceneView)
    {
        Ray ray = HandleUtility.GUIPointToWorldRay(mousePosition);
        Plane plane = new Plane(Vector3.forward, Vector3.zero);
        if (plane.Raycast(ray, out float enter))
            return ray.GetPoint(enter);
        return Vector3.positiveInfinity;
    }

        if (groundLayer >= 0)
            obj.layer = groundLayer;

그러면 Undo 했을 때:
- 부모만 지워지거나
- 자식이 남거나
- 상태가 꼬일 수 있습니다 (에디터 툴에서 제일 짜증나는 종류)

✅ 실전 조치
- `Undo.IncrementCurrentGroup()` / `Undo.SetCurrentGroupName()` / `Undo.CollapseUndoOperations()`로 그룹 처리
- 그리고 생성되는 세그먼트마다 `Undo.RegisterCreatedObjectUndo(seg, ...)` 등록

# 🚀 LÖVE2D로 “그대로” 가져갈 수 있는 것들

## 1) **두 트랙 구조**: 패턴 생성기 + 자유형 편집기
유니티 툴의 진짜 강점이 이거였습니다.

- **패턴 생성기**(LevelGenerator): “A-1/A-2/A-3” 같은 **프리셋 조합**
- **자유형 편집기**(SplineTerrainSet): 손으로 찍고 드래그하는 **노드 편집**

✅ LÖVE2D에선
- `prefabs/segments.lua` (패턴 라이브러리)
- `editor.lua` (노드 편집)
이렇게 두 갈래로 두면, 맵 제작 속도 확 올라갑니다.

---

## 2) **그룹(부모) 아래로 생성물 묶기** 개념
유니티에선 GameObject 그룹이었죠.  
LÖVE2D에선 데이터 구조로:

```lua
level = {
  groups = {
    {
      name="stage1A",
      items = { ...segments... },
      meta = { seed=..., tags=... }
    }
  }
}
```

이렇게 “그룹 단위”가 생기면:
- 삭제/이동/복제
- Undo/Redo
- 저장/로드
가 전부 쉬워집니다.

---

## 3) **재질/레이어/태그 자동 적용** (디버깅 확률 낮춤)
유니티에서 collider/material/layer 자동이 핵심이었는데, LÖVE에선 이렇게 대응됩니다:

- `mat`: `"sand" | "rock" | "no_spawn"`
- `collision`: `"ground" | "oneway" | "none"`
- `layer`: `"bg" | "mid" | "fg"`

→ 나중에 **발소리/먼지/AI 스폰 금지/카메라 패럴랙스**까지 한 번에 연결됩니다.

---

## 4) **Undo/Redo는 필수 사양**
유니티 툴에서 Undo가 약했던 게 리스크였는데,  
LÖVE 에디터에서는 오히려 **처음부터 제대로** 넣으면 됩니다.

룰은 1개:
- 드래그 중 매프레임 기록 ❌
- “행동 시작 1회” 스냅샷 ✅

이거 지키면 버그/메모리/성능 다 안정적입니다.

---

## 5) “Shift+클릭 배치” UX 자체를 이식
유니티에서 Shift+클릭으로 찍는 게 생산성 핵심이었죠.

LÖVE에서도 그대로:
- Shift+클릭: **세그먼트 프리셋 배치**
- 그냥 클릭/드래그: **지형 노드 편집**

이 UX는 몸에 익으면 못 버립니다.
