```python
#!/usr/bin/env python3
from __future__ import annotations

import argparse
import os
import re
import sys
from dataclasses import dataclass
from pathlib import Path
from typing import Iterable


@dataclass(frozen=True)
class Finding:
    path: Path
    line_no: int
    line: str
    rule: str


DEFAULT_EXCLUDE_DIRS = {
    ".git", ".venv", "venv", "__pycache__", ".pytest_cache", "node_modules",
    "dist", "build", ".idea", ".vscode",
}

# allowlist: docs and tooling can mention "EnemyAI hit" as text
DEFAULT_ALLOWLIST_PATH_PREFIXES = (
    "Docs/",
    "docs/",
    "tools/guards/",
    "tests/",
)

# Target scope: EnemyAI* scripts are the critical zone
ENEMYAI_FILE_HINTS = (
    "EnemyAI",
    "enemy_ai",
)

# Forbidden patterns: keep conservative; expand when incidents happen.
FORBIDDEN_PATTERNS: list[tuple[str, re.Pattern[str]]] = [
    ("enemyai_hit_keyword", re.compile(r"\bEnemyAI\b.*\bhit\b", re.IGNORECASE)),
    ("enemyai_onhit", re.compile(r"\bEnemyAI\b.*\bonhit\b", re.IGNORECASE)),
    ("enemyai_applydamage", re.compile(r"\bEnemyAI\b.*\b(apply|take)\s*damage\b", re.IGNORECASE)),
    # Also catch direct hit handlers inside EnemyAI files even without the "EnemyAI" token
    ("enemyai_file_hit_func", re.compile(r"\b(onhit|applydamage|takedamage|hit)\b", re.IGNORECASE)),
]


def _is_excluded_dir(p: Path, exclude_dirs: set[str]) -> bool:
    parts = set(p.parts)
    return any(d in parts for d in exclude_dirs)


def _is_allowlisted(path: Path, allow_prefixes: tuple[str, ...]) -> bool:
    s = path.as_posix()
    return s.startswith(allow_prefixes)


def iter_files(root: Path, exclude_dirs: set[str]) -> Iterable[Path]:
    for dirpath, dirnames, filenames in os.walk(root):
        dp = Path(dirpath)
        # prune excluded dirs
        dirnames[:] = [d for d in dirnames if d not in exclude_dirs]
        if _is_excluded_dir(dp, exclude_dirs):
            continue
        for fn in filenames:
            p = dp / fn
            # only scan text-like files
            if p.suffix.lower() in {".mlua", ".lua", ".py", ".md", ".txt"}:
                yield p


def scan_file(path: Path) -> list[Finding]:
    findings: list[Finding] = []
    try:
        text = path.read_text(encoding="utf-8", errors="replace").splitlines()
    except Exception as e:
        return [Finding(path, 0, f"<read error: {e}>", "read_error")]

    is_enemyai_file = any(h.lower() in path.name.lower() for h in ENEMYAI_FILE_HINTS)

    for i, line in enumerate(text, start=1):
        for rule, pat in FORBIDDEN_PATTERNS:
            # tighten rule for non-EnemyAI files: only trip rules that explicitly mention EnemyAI
            if (not is_enemyai_file) and rule == "enemyai_file_hit_func":
                continue
            if pat.search(line):
                findings.append(Finding(path, i, line.rstrip("\n"), rule))
    return findings


def main(argv: list[str]) -> int:
    ap = argparse.ArgumentParser(description="Guard: forbid EnemyAI hit logic (SSOT invariant)")
    ap.add_argument("--root", default=".", help="repo root to scan")
    ap.add_argument("--exclude-dir", action="append", default=[], help="exclude dir name (repeatable)")
    ap.add_argument("--allow-prefix", action="append", default=[], help="allowlisted path prefix (repeatable)")
    args = ap.parse_args(argv)

    root = Path(args.root).resolve()
    exclude_dirs = set(DEFAULT_EXCLUDE_DIRS) | set(args.exclude_dir)
    allow_prefixes = tuple(DEFAULT_ALLOWLIST_PATH_PREFIXES + tuple(args.allow_prefix))

    all_findings: list[Finding] = []

    for p in iter_files(root, exclude_dirs):
        rel = p.relative_to(root)
        if _is_allowlisted(rel, allow_prefixes):
            continue
        fs = scan_file(p)
        # filter out read_error only if it’s in binary-ish zone; keep as fail by default (debuggable)
        all_findings.extend(fs)

    bad = [f for f in all_findings if f.rule != "read_error"]
    read_errs = [f for f in all_findings if f.rule == "read_error"]

    if bad:
        print("[GUARD][FAIL] EnemyAI hit invariant violated.")
        for f in bad[:200]:
            print(f"- {f.path.as_posix()}:{f.line_no} [{f.rule}] {f.line}")
        if len(bad) > 200:
            print(f"... ({len(bad) - 200} more)")
        return 11

    if read_errs:
        print("[GUARD][FAIL] Read errors encountered while scanning (treat as failure).")
        for f in read_errs[:50]:
            print(f"- {f.path.as_posix()} [{f.rule}] {f.line}")
        return 12

    print("[GUARD][OK] forbid_enemyai_hit")
    return 0


if __name__ == "__main__":
    raise SystemExit(main(sys.argv[1:]))
```

```python
from __future__ import annotations

import subprocess
import sys
from pathlib import Path


def run_guard(root: Path) -> subprocess.CompletedProcess[str]:
    guard = Path("tools/guards/forbid_enemyai_hit.py")
    assert guard.exists(), "guard script missing"
    return subprocess.run(
        [sys.executable, str(guard), "--root", str(root)],
        text=True,
        capture_output=True,
    )


def test_guard_allows_non_enemyai_mentions(tmp_path: Path) -> None:
    # docs mention should be allowlisted by prefix; simulate a non-allowlisted md to ensure it fails only when needed
    (tmp_path / "SomeScript.mlua").write_text("log('hello')\n", encoding="utf-8")
    p = run_guard(tmp_path)
    assert p.returncode == 0, p.stdout + "\n" + p.stderr


def test_guard_blocks_enemyai_hit_in_enemyai_file(tmp_path: Path) -> None:
    (tmp_path / "EnemyAI_Test.mlua").write_text("function OnHit() end\n", encoding="utf-8")
    p = run_guard(tmp_path)
    assert p.returncode != 0
    assert "[GUARD][FAIL]" in p.stdout
```

```md
  - dbg payload 추가
- `release-check / pytest / smoke` 통과

## 파이프라인 개요(권장 구조)
1) **서버(ServerOnly)가 hit를 판정/기록**한다.  
2) 서버는 hit 이벤트를 **ClientHUD로 단방향 전달**한다(표시/오버레이/연출 전용).  
3) 클라는 hit 판정/데미지 확정 로직을 가지지 않는다(치트/권한 문제 방지).

## 불변식(Invariants)
- **EnemyAI는 hit를 처리하지 않는다.** (절대 금지)
- hit 판정/확정은 **서버에서만** 한다.
- 클라는 “표시/연출/디버그”만 담당한다.
- 디버그는 반드시 **오버레이 + dbg payload**로 관찰 가능해야 한다.

## 디버그 오버레이 규격
- HUD 상단(또는 지정 영역)에 최근 hit N개를 리스트로 표시
- 각 항목은 최소 다음을 포함:
  - time(t)
  - attacker/target 식별자(가능하면)
  - dmg/crit 등 결과
  - dbg payload(원본/요약)

## 흔한 실수 / 금지 사례
- EnemyAI에서 onHit/TakeDamage/ApplyDamage 류 로직을 다시 추가하는 것
- 클라에서 데미지 확정/HP 변경을 수행하는 것
- dbg payload 없이 로그만 남겨 “재현 불가” 상태가 되는 것
```

```python
#!/usr/bin/env python3
from __future__ import annotations

import argparse
import os
import re
import sys
from dataclasses import dataclass
from pathlib import Path
from typing import Iterable


@dataclass(frozen=True)
class Finding:
    path: Path
    line_no: int
    line: str
    rule: str


DEFAULT_EXCLUDE_DIRS = {
    ".git", ".venv", "venv", "__pycache__", ".pytest_cache", "node_modules",
    "dist", "build", ".idea", ".vscode",
}

DEFAULT_ALLOWLIST_PATH_PREFIXES = (
    "Docs/",
    "docs/",
    "tools/guards/",
    "tests/",
)

ENEMYAI_FILE_HINTS = (
    "EnemyAI",
    "enemy_ai",
)

FORBIDDEN_CALL_SYMBOLS = (
    "ApplyDamage",
    "ConsumeCombatEvent",
    "SendCombatEventToUser",
)

FORBIDDEN_ENEMYAI_HANDLER_PATTERNS: list[tuple[str, re.Pattern[str]]] = [
    (
        "enemyai_forbidden_function_decl",
        re.compile(
            r"\bfunction\s+(?:[A-Za-z_][A-Za-z0-9_]*[:.])?"
            r"(onhit|applydamage|takedamage|consumecombatevent|sendcombateventtouser)\s*\(",
            re.IGNORECASE,
        ),
    ),
    (
        "enemyai_forbidden_method_decl",
        re.compile(
            r"\bmethod\s+[A-Za-z_][A-Za-z0-9_]*\s+"
            r"(onhit|applydamage|takedamage|consumecombatevent|sendcombateventtouser)\s*\(",
            re.IGNORECASE,
        ),
    ),
]


def _is_excluded_dir(p: Path, exclude_dirs: set[str]) -> bool:
    parts = set(p.parts)
    return any(d in parts for d in exclude_dirs)


def _is_allowlisted(path: Path, allow_prefixes: tuple[str, ...]) -> bool:
    s = path.as_posix()
    return s.startswith(allow_prefixes)


def iter_files(root: Path, exclude_dirs: set[str]) -> Iterable[Path]:
    for dirpath, dirnames, filenames in os.walk(root):
        dp = Path(dirpath)
        dirnames[:] = [d for d in dirnames if d not in exclude_dirs]
        if _is_excluded_dir(dp, exclude_dirs):
            continue
        for fn in filenames:
            p = dp / fn
            if p.suffix.lower() in {".mlua", ".lua", ".py", ".md", ".txt"}:
                yield p


def scan_file(path: Path) -> list[Finding]:
    findings: list[Finding] = []
    try:
        text = path.read_text(encoding="utf-8", errors="replace").splitlines()
    except Exception as e:
        return [Finding(path, 0, f"<read error: {e}>", "read_error")]

    is_enemyai_file = any(h.lower() in path.name.lower() for h in ENEMYAI_FILE_HINTS)
    if not is_enemyai_file:
        return findings

    for i, line in enumerate(text, start=1):
        code_line = re.sub(r"--.*$", "", line)
        if not code_line.strip():
            continue

        for sym in FORBIDDEN_CALL_SYMBOLS:
            if re.search(rf"(\:|\.)\s*{re.escape(sym)}\s*\(", code_line) or re.search(
                rf"\b{re.escape(sym)}\s*\(",
                code_line,
            ):
                findings.append(Finding(path, i, line.rstrip("\n"), f"enemyai_forbidden_call:{sym}"))

        for rule, pat in FORBIDDEN_ENEMYAI_HANDLER_PATTERNS:
            if pat.search(line):
                findings.append(Finding(path, i, line.rstrip("\n"), rule))
    return findings


def main(argv: list[str]) -> int:
    ap = argparse.ArgumentParser(description="Guard: forbid EnemyAI hit logic (SSOT invariant)")
    ap.add_argument("--root", default=".", help="repo root to scan")
    ap.add_argument("--exclude-dir", action="append", default=[], help="exclude dir name (repeatable)")
    ap.add_argument("--allow-prefix", action="append", default=[], help="allowlisted path prefix (repeatable)")
    args = ap.parse_args(argv)

    root = Path(args.root).resolve()
    exclude_dirs = set(DEFAULT_EXCLUDE_DIRS) | set(args.exclude_dir)
    allow_prefixes = tuple(DEFAULT_ALLOWLIST_PATH_PREFIXES + tuple(args.allow_prefix))

    all_findings: list[Finding] = []

    for p in iter_files(root, exclude_dirs):
        rel = p.relative_to(root)
        if _is_allowlisted(rel, allow_prefixes):
            continue
        fs = scan_file(p)
        all_findings.extend(fs)

    bad = [f for f in all_findings if f.rule != "read_error"]
    read_errs = [f for f in all_findings if f.rule == "read_error"]

    if bad:
        print("[GUARD][FAIL] EnemyAI hit invariant violated.")
        for f in bad[:200]:
            print(f"- {f.path.as_posix()}:{f.line_no} [{f.rule}] {f.line}")
        if len(bad) > 200:
            print(f"... ({len(bad) - 200} more)")
        return 11

    if read_errs:
        print("[GUARD][FAIL] Read errors encountered while scanning (treat as failure).")
        for f in read_errs[:50]:
            print(f"- {f.path.as_posix()} [{f.rule}] {f.line}")
        return 12

    print("[GUARD][OK] forbid_enemyai_hit")
    return 0


if __name__ == "__main__":
    raise SystemExit(main(sys.argv[1:]))
```

```python
#!/usr/bin/env python3
from __future__ import annotations

import argparse
import re
import sys
from pathlib import Path


EXIT_MISSING = 21
EXIT_MISMATCH = 22

SSOT_PATH_DEFAULT = "Docs/SSOT/Stage2_Hit_Pipeline.md"
CODE_CONST_PATH_DEFAULT = "tools/guards/hit_dbg_payload_v1_fields.txt"
TEST_CONST_PATH_DEFAULT = "tests/hit_dbg_payload_v1_fields.txt"


def _read_text(p: Path) -> str:
    return p.read_text(encoding="utf-8", errors="replace")


def parse_fields_from_ssot(md: str) -> set[str]:
    # Extract only the bullet list under "### v1 fields"
    m = re.search(r"###\s+v1 fields.*?\n(.*?)(?:\n###|\Z)", md, flags=re.IGNORECASE | re.DOTALL)
    if not m:
        return set()
    block = m.group(1)
    fields: set[str] = set()
    for line in block.splitlines():
        line = line.strip()
        if not line.startswith("-"):
            continue
        # "- `name`:" pattern
        m2 = re.search(r"`([A-Za-z0-9_]+)`", line)
        if m2:
            fields.add(m2.group(1))
    return fields


def parse_fields_from_list_text(txt: str) -> set[str]:
    fields: set[str] = set()
    for line in txt.splitlines():
        s = line.strip()
        if not s or s.startswith("#"):
            continue
        fields.add(s)
    return fields


def main(argv: list[str]) -> int:
    ap = argparse.ArgumentParser(description="Guard: dbg payload v1 schema gate (SSOT+guard+test must match)")
    ap.add_argument("--root", default=".", help="repo root")
    ap.add_argument("--ssot", default=SSOT_PATH_DEFAULT, help="SSOT md path")
    ap.add_argument("--guard-fields", default=CODE_CONST_PATH_DEFAULT, help="guard fields list path")
    ap.add_argument("--test-fields", default=TEST_CONST_PATH_DEFAULT, help="test fields list path")
    args = ap.parse_args(argv)

    root = Path(args.root).resolve()
    ssot_p = (root / args.ssot)
    guard_p = (root / args.guard_fields)
    test_p = (root / args.test_fields)

    missing = [p for p in (ssot_p, guard_p, test_p) if not p.exists()]
    if missing:
        print("[GUARD][FAIL] dbg payload v1 gate: missing required files:")
        for p in missing:
            print(f"- {p.as_posix()}")
        return EXIT_MISSING

    ssot_fields = parse_fields_from_ssot(_read_text(ssot_p))
    guard_fields = parse_fields_from_list_text(_read_text(guard_p))
    test_fields = parse_fields_from_list_text(_read_text(test_p))

    if not ssot_fields:
        print("[GUARD][FAIL] dbg payload v1 gate: could not parse SSOT v1 fields block")
        return EXIT_MISSING

    if ssot_fields != guard_fields or ssot_fields != test_fields:
        print("[GUARD][FAIL] dbg payload v1 gate: mismatch detected")
        print(f"- SSOT : {sorted(ssot_fields)}")
        print(f"- GUARD: {sorted(guard_fields)}")
        print(f"- TEST : {sorted(test_fields)}")
        print("Fix: update SSOT + tools/guards/hit_dbg_payload_v1_fields.txt + tests/hit_dbg_payload_v1_fields.txt together.")
        return EXIT_MISMATCH

    print("[GUARD][OK] dbg payload v1 gate")
    return 0


if __name__ == "__main__":
    raise SystemExit(main(sys.argv[1:]))
```

```mlua

self.xp = self.xp or 0
self.lv = self.lv or 1

self.atkCdMul = self.atkCdMul or 1.0

```

```mlua
-- OnUpdate(dt) 내부(서버)
self.t = (self.t or 0) + dt

-- spawn
if self.spawnEnabled then
    self._spawnAcc = self._spawnAcc + dt
    if self._spawnAcc >= self.spawnPeriod then
        self._spawnAcc = 0
        if self:_CountAliveEnemies() < self.spawnCap then
            self:_SpawnEnemyNearPlayer()
        end
    end
end

-- auto attack (single target nearest)
if self.atkEnabled and (not self._pendingLevelUp) then
    self._atkAcc = self._atkAcc + dt
    local period = (self.atkPeriod * self.atkCdMul)
    if self._atkAcc >= period then
        self._atkAcc = 0
        local enemy = self:_FindNearestEnemy()
        if enemy ~= nil then
            local dmg = math.floor(10 * self.damageMul)
            self:_Server_ApplyHit(enemy, dmg) -- 반드시 서버에서 확정(기존 hit 파이프라인 사용)
        end
    end
end

-- level up trigger
while self.xp >= self.xpToNext do
    self.xp = self.xp - self.xpToNext
    self.lv = self.lv + 1
    self.xpToNext = math.floor(self.xpToNext * 1.25)
    self._pendingLevelUp = true
    self:Client_ShowLevelUpChoices(self.lv, self:_RollChoices3()) -- HUD에 3개 텍스트
end

-- HUD sync (이미 있던 방식 그대로)
self:Client_ReceivePlayerState(self.hp, self.xp, self.lv, self.t)
```

```mlua
-- RPC: 클라 선택 수신(서버)
@ExecSpace("ServerOnly")
method void Server_ChooseUpgrade(string choiceId)
    if not self._pendingLevelUp then return end
    self._pendingLevelUp = false

    if choiceId == "UPG_ATK_CD" then
        self.atkCdMul = self.atkCdMul * 0.9
    elseif choiceId == "UPG_DMG" then
        self.damageMul = self.damageMul * 1.1
    elseif choiceId == "UPG_HP" then
        self.hp = math.min(self.hp + 20, 200)
    end
end
```

```mlua
property boolean overlayEnabled = true
property integer overlayHz = 10
property number _overlayAccumSec = 0

@ExecSpace("ClientOnly")
method void _RenderDebugOverlayWithThrottle(table snapshot, boolean forceRender)
    if self.overlayEnabled ~= true then return end
    if self.overlayHz == nil then self.overlayHz = 10 end
    if self.overlayHz <= 0 then return end

    if forceRender == true then
        self._overlayAccumSec = 0
        self:RenderDebugOverlay(snapshot)
        return
    end

    local hz = tonumber(self.overlayHz) or 10
    local period = 1.0 / hz
    self._overlayAccumSec = (tonumber(self._overlayAccumSec) or 0) + 0.1
    if self._overlayAccumSec < period then return end

    self._overlayAccumSec = 0
    self:RenderDebugOverlay(snapshot)
end
```
